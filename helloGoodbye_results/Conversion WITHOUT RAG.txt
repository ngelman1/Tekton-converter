=== Conversion WITHOUT RAG ===
```yaml
apiVersion: tekton.dev/v1
kind: Task
metadata:
  name: create-greeting
spec:
  workspaces:
    - name: shared-data
      description: The workspace to write the initial message to.
  steps:
    - name: create
      image: bash:5.2
      script: |
        #!/usr/bin/env bash
        set -e
        MESSAGE="Hello from Stage 1!"
        echo -n "${MESSAGE}" > $(workspaces.shared-data.path)/message.txt
        echo "Stage 1 Message: ${MESSAGE}"
---


apiVersion: tekton.dev/v1
kind: Task
metadata:
  name: append-to-greeting
spec:
  workspaces:
    - name: shared-data
      description: The workspace to read and append the message to.
  steps:
    - name: append
      image: bash:5.2
      script: |
        #!/usr/bin/env bash
        set -e
        MESSAGE_FILE=$(workspaces.shared-data.path)/message.txt
        echo -n " Goodbye from Stage 2!" >> "${MESSAGE_FILE}"
        FULL_MESSAGE=$(cat "${MESSAGE_FILE}")
        echo "Stage 2 Message: ${FULL_MESSAGE}"
---


apiVersion: tekton.dev/v1
kind: Task
metadata:
  name: print-final-message
spec:
  workspaces:
    - name: shared-data
      description: The workspace to read the final message from.
  steps:
    - name: print
      image: bash:5.2
      script: |
        #!/usr/bin/env bash
        set -e
        MESSAGE_FILE=$(workspaces.shared-data.path)/message.txt
        if [[ -f "${MESSAGE_FILE}" ]]; then
          echo "Final Message: $(cat ${MESSAGE_FILE})"
        else
          echo "Message file not found. Previous steps may have failed."
        fi
---


apiVersion: tekton.dev/v1
kind: Pipeline
metadata:
  name: shared-variable-pipeline
spec:
  workspaces:
    - name: pipeline-shared-data
      description: Shared data across pipeline tasks.
  tasks:
    - name: create-message
      taskRef:
        name: create-greeting
      workspaces:
        - name: shared-data
          workspace: pipeline-shared-data
    - name: append-to-message
      taskRef:
        name: append-to-greeting
      runAfter:
        - create-message
      workspaces:
        - name: shared-data
          workspace: pipeline-shared-data
  finally:
    - name: print-the-message
      taskRef:
        name: print-final-message
      workspaces:
        - name: shared-data
          workspace: pipeline-shared-data
---


apiVersion: tekton.dev/v1
kind: PipelineRun
metadata:
  name: shared-variable-pipelinerun
spec:
  pipelineRef:
    name: shared-variable-pipeline
  workspaces:
    - name: pipeline-shared-data
      emptyDir: {}
```
